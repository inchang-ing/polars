20/// Table functions that are supported by Polars
21#[allow(clippy::enum_variant_names)]
22pub(crate) enum PolarsTableFunctions {
23    /// SQL 'read_csv' function, with optional named options as in Python's `scan_csv`.
24    /// ```sql
25    /// SELECT * FROM read_csv('path/to/file.csv')
26    /// SELECT * FROM read_csv('path/to/file.csv', has_header => false, separator => ';')
27    /// ```
28    #[cfg(feature = "csv")]
29    ReadCsv,
30    /// SQL 'read_parquet' function.
31    /// ```sql
32    /// SELECT * FROM read_parquet('path/to/file.parquet')
33    /// ```
34    #[cfg(feature = "parquet")]
35    ReadParquet,
36    /// SQL 'read_ipc' function.
37    /// ```sql
38    /// SELECT * FROM read_ipc('path/to/file.ipc')
39    /// ```
40    #[cfg(feature = "ipc")]
41    ReadIpc,
42    /// SQL 'read_json' function (*only ndjson is currently supported*).
43    /// ```sql
44    /// SELECT * FROM read_json('path/to/file.json')
45    /// ```
46    #[cfg(feature = "json")]
47    ReadJson,
48}
49
50impl FromStr for PolarsTableFunctions {
51    type Err = PolarsError;
52
53    #[allow(unreachable_code)]
54    fn from_str(s: &str) -> Result<Self, Self::Err> {
55        Ok(match s.to_lowercase().as_str() {
56            #[cfg(feature = "csv")]
57            "read_csv" => PolarsTableFunctions::ReadCsv,
58            #[cfg(feature = "parquet")]
59            "read_parquet" => PolarsTableFunctions::ReadParquet,
60            #[cfg(feature = "ipc")]
61            "read_ipc" => PolarsTableFunctions::ReadIpc,
62            #[cfg(feature = "json")]
63            "read_json" => PolarsTableFunctions::ReadJson,
64            _ => polars_bail!(SQLInterface: "'{}' is not a supported table function", s),
65        })
66    }
67}

69impl PolarsTableFunctions {
70    #[allow(unused_variables, unreachable_patterns)]
71    pub(crate) fn execute(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
72        match self {
73            #[cfg(feature = "csv")]
74            PolarsTableFunctions::ReadCsv => self.read_csv(args),
75            #[cfg(feature = "parquet")]
76            PolarsTableFunctions::ReadParquet => self.read_parquet(args),
77            #[cfg(feature = "ipc")]
78            PolarsTableFunctions::ReadIpc => self.read_ipc(args),
79            #[cfg(feature = "json")]
80            PolarsTableFunctions::ReadJson => self.read_ndjson(args),
81            _ => unreachable!(),
82        }
83    }
84
85    #[cfg(feature = "csv")]
86    fn read_csv(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
87        use std::sync::Arc;
88
89        use polars_io::csv::read::{CsvEncoding, NullValues};
90        use polars_lazy::frame::LazyFileListReader;
91
92        let (path_arg, options) = TableFunctionOptions::parse(
93            "read_csv",
94            args,
95            READ_CSV_OPTIONS,
96            READ_CSV_EXCLUDED_OPTIONS,
97        )?;
98        let path = self.get_file_path_from_arg(path_arg)?;
99
100        let has_header = options.bool("has_header")?.unwrap_or(true);
101        let schema = options.schema("schema")?;
102        let mut reader = LazyCsvReader::new(path.clone())
103            .with_has_header(has_header)
104            .with_try_parse_dates(options.bool("try_parse_dates")?.unwrap_or(true))
105            // As in Python: an empty file only errors when there is no schema to return.
106            .with_raise_if_empty(schema.is_none() || has_header)
107            .with_schema(schema.map(Arc::new));
108
109        if let Some(separator) = options.byte("separator")? {
110            reader = reader.with_separator(separator);
111        }
112        if let Some(quote_char) = options.nullable_byte("quote_char")? {
113            reader = reader.with_quote_char(quote_char);
114        }
115        if let Some(prefix) = options.nullable_string("comment_prefix")? {
116            reader = reader.with_comment_prefix(prefix.map(PlSmallStr::from));
117        }
118        if let Some(n) = options.usize("skip_rows")? {
119            reader = reader.with_skip_rows(n);
120        }
121        if let Some(n) = options.usize("skip_rows_after_header")? {
122            reader = reader.with_skip_rows_after_header(n);
123        }
124        if let Some(n) = options.usize("skip_lines")? {
125            reader = reader.with_skip_lines(n);
126        }
127        if let Some(encoding) = options.string("encoding")? {
128            reader = reader.with_encoding(match encoding.to_lowercase().as_str() {
129                "utf8" => CsvEncoding::Utf8,
130                "utf8-lossy" => CsvEncoding::LossyUtf8,
131                _ => polars_bail!(
132                    SQLSyntax: "`read_csv` option 'encoding' expects 'utf8' or 'utf8-lossy'; found: '{}'", encoding
133                ),
134            });
135        }
136        if let Some(names) = options.strings("new_columns")? {
137            let names: Vec<PlSmallStr> = names.into_iter().map(PlSmallStr::from).collect();
138            reader = reader.with_column_names_overwrite(names.into());
139        }
140        match options.dtype_overrides("schema_overrides")? {
141            Some(DtypeOverrides::ByName(schema)) => {
142                reader = reader.with_dtype_overwrite(Some(Arc::new(schema)));
143            },
144            Some(DtypeOverrides::ByPosition(dtypes)) => {
145                reader = reader.with_dtype_overwrite_by_position(Some(Arc::new(dtypes)));
146            },
147            None => {},
148        }
149        if let Some(n) = options.nullable_usize("infer_schema_length")? {
150            reader = reader.with_infer_schema_length(n);
151        }
152        // As in Python: `infer_schema => false` reads every column as a string.
153        if options.bool("infer_schema")? == Some(false) {
154            reader = reader.with_infer_schema_length(Some(0));
155        }
156        if let Some(decimal_comma) = options.bool("decimal_comma")? {
157            reader = reader.with_decimal_comma(decimal_comma);
158        }
159        let null_values = options
160            .string_values("null_values")?
161            .map(|values| match values {
162                StringValues::One(value) => NullValues::AllColumnsSingle(value.into()),
163                StringValues::List(values) => {
164                    NullValues::AllColumns(values.into_iter().map(Into::into).collect())
165                },
166                StringValues::Dict(pairs) => NullValues::Named(
167                    pairs
168                        .into_iter()
169                        .map(|(k, v)| (k.into(), v.into()))
170                        .collect(),
171                ),
172            });
173        reader = reader.with_null_values(null_values);
174        if let Some(ignore_errors) = options.bool("ignore_errors")? {
175            reader = reader.with_ignore_errors(ignore_errors);
176        }
177        if let Some(truncate) = options.bool("truncate_ragged_lines")? {
178            reader = reader.with_truncate_ragged_lines(truncate);
179        }
180        if let Some(column) = options.string("include_file_paths")? {
181            reader = reader.with_include_file_paths(Some(column.into()));
182        }
183
184        let lf = reader.finish()?;
185        Ok((path, lf))
186    }

188    #[cfg(feature = "parquet")]
189    fn read_parquet(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
190        polars_ensure!(args.len() == 1, SQLSyntax: "`read_parquet` expects a single file path; found {:?} arguments", args.len());
191
192        let path = self.get_file_path_from_arg(&args[0])?;
193        let lf = LazyFrame::scan_parquet(path.clone(), Default::default())?;
194        Ok((path, lf))
195    }
196
197    #[cfg(feature = "ipc")]
198    fn read_ipc(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
199        polars_ensure!(args.len() == 1, SQLSyntax: "`read_ipc` expects a single file path; found {:?} arguments", args.len());
200
201        let path = self.get_file_path_from_arg(&args[0])?;
202        let lf = LazyFrame::scan_ipc(path.clone(), Default::default(), Default::default())?;
203        Ok((path, lf))
204    }
205    #[cfg(feature = "json")]
206    fn read_ndjson(&self, args: &[SQLFunctionArg]) -> PolarsResult<(PlRefPath, LazyFrame)> {
207        polars_ensure!(args.len() == 1, SQLSyntax: "`read_ndjson` expects a single file path; found {:?} arguments", args.len());
208
209        use polars_lazy::frame::LazyFileListReader;
210        use polars_lazy::prelude::LazyJsonLineReader;
211
212        let path = self.get_file_path_from_arg(&args[0])?;
213        let lf = LazyJsonLineReader::new(path.clone()).finish()?;
214        Ok((path, lf))
215    }
216
217    #[allow(dead_code)]
218    fn get_file_path_from_arg(&self, arg: &SQLFunctionArg) -> PolarsResult<PlRefPath> {
219        match arg {
220            SQLFunctionArg::Unnamed(SQLFunctionArgExpr::Expr(SQLExpr::Value(
221                SQLValueWithSpan {
222                    value: SQLValue::SingleQuotedString(s) | SQLValue::EscapedStringLiteral(s),
223                    ..
224                },
225            ))) => Ok(PlRefPath::new(s)),
226            _ => polars_bail!(
227                SQLSyntax:
228                "expected a valid file path as a single-quoted string; found: {}", arg,
229            ),
230        }
231    }
232}
